{{- /* Tri-state patch indicator. Context (.) is a dict: page — the page style — "text" (default), "dot", or "chip" patch_status is set by build-hugo-content.py from an authoritative fix record (OSV / NVD) where one exists, and only falls back to reading the README when no source knows the CVE. "unknown" is a real answer here — it is not the same as "unpatched". */ -}} {{- $p := .page -}} {{- $style := .style | default "text" -}} {{- $s := $p.Params.patch_status | default (cond $p.Params.patched "patched" "unknown") -}} {{- $label := "Unverified" -}}{{- $cls := "text-zinc-500 dark:text-mono-muted" -}}{{- $dot := "bg-zinc-400" -}} {{- if eq $s "patched" }}{{ $label = "Patched" }}{{ $cls = "text-emerald-600 dark:text-brand" }}{{ $dot = "bg-emerald-500" }} {{- else if eq $s "unpatched" }}{{ $label = "Unpatched" }}{{ $cls = "text-red-600 dark:text-red-400" }}{{ $dot = "bg-red-500" }}{{ end -}} {{- $title := "" -}} {{- if eq $s "patched" }} {{- if $p.Params.fix_source }}{{ $title = printf "Fix confirmed via %s%s" (upper $p.Params.fix_source) (cond (ne (string ($p.Params.fix_detail | default "")) "") (printf ": %s" $p.Params.fix_detail) "") }} {{- else }}{{ $title = "Vendor fix documented in this entry" }}{{ end }} {{- else if eq $s "unpatched" }}{{ $title = "No vendor fix at the time of writing" }} {{- else }}{{ $title = "No authoritative fix record found; status not confirmed either way" }}{{ end -}} {{- if eq $style "dot" -}} {{ $label }} {{- else if eq $style "chip" -}} {{ $label }} {{- else -}} {{ $label }} {{- end -}}
